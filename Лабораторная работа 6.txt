#define _CRT_SECURE_NO_DEPRECATE
#define _USE_MATH_DEFINES 
#include <locale.h>
#include <stdio.h>
#include <math.h>
int z_1(int years);
double z_2(double x);

int main()
{
    setlocale(LC_ALL, "RUS"); 
    
    int years, res;
    puts("Введите год");
    scanf("%d", &years);
    res= z_1(years);
    if (res) {
		printf("Год %d високосный\n", years);
	}
	else {
		printf("Год %d не високосный\n", years);
	}
	
	double x, result;
	puts("\nВведите число");
	scanf("%lf", &x);
	result = z_2(x);
	printf("F(%lf) = %.3f, использована формула %s\n", x, result, (x <= 7) ? "3x - 9, при x <= 7" : "1/(x^2 - 4), при x > 7");
    
    return 0;
}

int z_1(int years)
{
    int r1;
    r1 = ((years % 4 == 0 && years % 100 != 0) || (years % 400 == 0));
    return r1;
}

double z_2(double x)
{   //Вариант 6
    //       | 3x - 9, при x <= 7
    //F(x) = {
    //       |1/(x^2 - 4), при x > 7
    //
    //Контрольные примеры:
    //1) x = 0 -> 0 <= 7 -> F(0) = 3*0 - 9 = -9
    //2) x = 1 -> 1 <= 7 -> F(1) = 3*1 - 9 = -6
    //3) x = 3 -> 3 <= 7 -> F(3) = 3*3 - 9 = 0
    //4) x = 8 -> 8 > 7 -> F(8) = 1/(8^2 - 4) = 1/60 = 0.017
    //5) x = 9 -> 9 > 7 -> F(9) = 1/(9^2 - 4) = 1/77 = 0.013
    //6) x = 11 -> 11 > 7 -> F(11) = 1/(11^2 - 4) = 1/117 = 0.009
    
    double F;
    if (x <= 7) {
		F = 3 * x - 9;
	}
	else {
		F = 1. / (pow(x, 2.) - 4);
	}
    return F;
}
